Kumo Tooling: CI/CD, Changesets, PR Reporters, and Visual Regression

Kumo's CI system includes changeset-based versioning, npm release automation, visual regression testing, and PR comment reporters that consolidate results from multiple CI jobs.

Source: ci/ (reporters, scripts, utils, visual-regression), tools/deployments/.

Reporter Registry

// ci/reporters/index.ts
import { npmReleaseReporter } from "./npm-release"
import { kumoDocsPreviewReporter } from "./kumo-docs-preview"
import { visualRegressionReporter } from "./visual-regression"

export const reporters: Reporter[] = [
  npmReleaseReporter,
  kumoDocsPreviewReporter,
  visualRegressionReporter,
]

Three reporters, sorted by priority in the consolidated PR comment:

  1. npm-release (priority 10-19) — Release info, version, changelog
  2. kumo-docs-preview (priority 20-29) — Docs preview URL
  3. visual-regression (priority 30-39) — Visual diff results

ReportItem Type

// ci/reporters/types.ts
export interface ReportItem {
  id: string       // e.g., "npm-release"
  title: string    // Section title in PR comment
  priority: number // Sort order (lower = first)
  content: string  // Markdown content
  success: boolean // Whether this job succeeded
}

GitHub API Integration

Source: ci/utils/github-api.ts

export const GITHUB_API_URL = "https://api.github.com"
export const GITHUB_REPO_OWNER = "cloudflare"
export const GITHUB_REPO_NAME = "kumo"

export async function createPullRequest(token, options) {
  const octokit = new Octokit({ auth: token })
  return octokit.pulls.create({
    owner: GITHUB_REPO_OWNER,
    repo: GITHUB_REPO_NAME,
    head: options.sourceBranch,
    base: options.targetBranch,
    title: options.title,
    body: options.description,
  })
}

export async function postPRComment(token, prNumber, body) {
  const octokit = new Octokit({ auth: token })
  return octokit.issues.createComment({
    owner: GITHUB_REPO_OWNER,
    repo: GITHUB_REPO_NAME,
    issue_number: prNumber,
    body,
  })
}
